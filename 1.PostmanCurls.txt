'

Postman Test Script (Assertions in JavaScript)
pm.test("Status code is 200", function () {

2. POST-negative testcase1
curl --location 'https://reqres.in/api/login' \
--header 'Content-Type: application/json' \
--data-raw '{
    "email": "eve.holt@reqres.in"
}
'


expected response:-
{
    "error": "Missing password"
}

pm.test("Status code is 400", function () {
    pm.response.to.have.status(400);
});

pm.test("Error message should indicate missing password", function () {
    var jsonData = pm.response.json();
    pm.expect(jsonData.error).to.eql("Missing password");
});
