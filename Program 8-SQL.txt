1.Create Table using SQL DDL Command
CREATE TABLE EMP_Data (
    EMP_No INT PRIMARY KEY,
    EMP_Name VARCHAR(100) NOT NULL,
    EMP_Exp INT NOT NULL,
    EMP_Department VARCHAR(100) NOT NULL,
    EMP_Salary INT NOT NULL
);

3.Query to Fetch Employees Working in the Same Department
SELECT *
FROM EMP_Data
WHERE EMP_Department IN (
    SELECT EMP_Department
    FROM EMP_Data
    GROUP BY EMP_Department
    HAVING COUNT(*) > 1
)
ORDER BY EMP_Department, EMP_No;
